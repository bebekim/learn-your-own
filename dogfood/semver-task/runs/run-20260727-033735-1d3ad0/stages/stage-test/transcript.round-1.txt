// ---------------------------------------------------------------------------
// Major / minor / patch numeric comparison

test('major version compared numerically', () => {
  assert.equal(compareSemver('1.0.0', '2.0.0'), -1);
  assert.equal(compareSemver('2.0.0', '1.0.0'), 1);
  assert.equal(compareSemver('1.0.0', '1.0.0'), 0);
});

test('multi-digit major versions compared numerically not lexically', () => {
  assert.equal(compareSemver('2.0.0', '10.0.0'), -1);
  assert.equal(compareSemver('10.0.0', '2.0.0'), 1);
  assert.equal(compareSemver('9.0.0', '10.0.0'), -1);
});

test('minor version compared numerically when major equal', () => {
  assert.equal(compareSemver('1.2.0', '1.3.0'), -1);
  assert.equal(compareSemver('1.3.0', '1.2.0'), 1);
  assert.equal(compareSemver('1.10.0', '1.2.0'), 1);
  assert.equal(compareSemver('1.2.0', '1.10.0'), -1);
});

test('patch version compared numerically when major and minor equal', () => {
  assert.equal(compareSemver('1.2.3', '1.2.4'), -1);
  assert.equal(compareSemver('1.2.4', '1.2.3'), 1);
  assert.equal(compareSemver('1.2.3', '1.2.3'), 0);
  assert.equal(compareSemver('1.2.10', '1.2.2'), 1);
  assert.equal(compareSemver('1.2.2', '1.2.10'), -1);
});

test('full numeric precedence chain: major then minor then patch', () => {

test('complex ordering chain is transitive-consistent', () => {
  // A < B < C < D, verify each adjacent pair and extremes
  const ordered = [
    '1.0.0-alpha',
    '1.0.0-alpha.1',
    '1.0.0-alpha.beta',
    '1.0.0-beta',
    '1.0.0-beta.2',
    '1.0.0-beta.11',
    '1.0.0-rc.1',
    '1.0.0',
  ];
  for (let i = 0; i < ordered.length - 1; i++) {
    assert.equal(compareSemver(ordered[i], ordered[i + 1]), -1,
      `${ordered[i]} < ${ordered[i + 1]}`);
  }
  assert.equal(compareSemver(ordered[0], ordered[ordered.length - 1]), -1);
  assert.equal(compareSemver(ordered[ordered.length - 1], ordered[0]), 1);
});
