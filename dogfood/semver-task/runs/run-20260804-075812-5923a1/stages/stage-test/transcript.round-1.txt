const test = require('node:test');
const assert = require('node:assert/strict');
const { compareSemver } = require('../src/semver.js');

test('compareSemver(a, b) === -compareSemver(b, a)', () => {
  const pairs = [
    ['1.0.0', '1.0.1'],
    ['1.0.0-alpha', '1.0.0'],
    ['1.0.0+build.5', '1.0.0'],
    ['1.0.1', '1.0.0'],
    ['1.0.0', '1.0.0-alpha'],
    ['1.0.0', '1.0.0+build.5'],
  ];
  for (const [a, b] of pairs) {
    assert.equal(
      (compareSemver(a, b) + 0),
      -(compareSemver(b, a) + 0)
    );
  }
});

test('compareSemver(a, a) === 0', () => {
  const versions = [
    '1.0.0',
    '1.0.0-alpha',
    '1.0.0-beta',
    '1.0.0-rc.1',
    '2.0.0',
    '10.0.0',
    '1.2.10',
    '1.0.0-alpha.1',
    '1.0.0-alpha.beta',
    '1.0.0-beta.2',
    '1.0.0-beta.11',
    '1.0.0+build.5',
    '1.0.0+build.10',
    '1.0.0+build.10+metadata',
    '1.0.0-alpha+build.5',
    '1.0.0-alpha.1+build.5',
    '1.0.0-alpha.1+build.10',
    '1.0.0-alpha.beta+build.5',
  ];
  for (const version of versions) {
    assert.equal(compareSemver(version, version) + 0, 0 + 0);
  }
