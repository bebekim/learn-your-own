'use strict';

function parseVersion(version) {
  // Strip build metadata
  var plusIndex = version.indexOf('+');
  var core = plusIndex === -1 ? version : version.substring(0, plusIndex);

  var prerelease = null;
  var dashIndex = core.indexOf('-');
  if (dashIndex !== -1) {
    prerelease = core.substring(dashIndex + 1);
    core = core.substring(0, dashIndex);
  }

  var parts = core.split('.');
  var major = parseInt(parts[0], 10);
  var minor = parseInt(parts[1], 10);
  var patch = parseInt(parts[2], 10);

  var prereleaseIdentifiers = null;
  if (prerelease !== null) {
    prereleaseIdentifiers = prerelease.split('.');
  }

  return {
    major: major,
    minor: minor,
    patch: patch,
    prereleaseIdentifiers: prereleaseIdentifiers
  };
}

function isNumericIdentifier(identifier) {
  return /^[0-9]+$/.test(identifier);
}

function comparePrereleaseIdentifiers(a, b) {
  var aIsNumeric = isNumericIdentifier(a);
  var bIsNumeric = isNumericIdentifier(b);

  if (aIsNumeric && bIsNumeric) {
    var aNum = parseInt(a, 10);
    var bNum = parseInt(b, 10);
    if (aNum < bNum) return -1;
    if (aNum > bNum) return 1;
    return 0;
  }

  if (aIsNumeric && !bIsNumeric) {
    return -1;
  }

  if (!aIsNumeric && bIsNumeric) {
    return 1;
  }

  // Both alphanumeric: lexical ASCII comparison
  if (a < b) return -1;
  if (a > b) return 1;
  return 0;
}

function compareSemver(a, b) {
  var va = parseVersion(a);
  var vb = parseVersion(b);

  // Compare major, minor, patch numerically
  if (va.major !== vb.major) {
    return va.major < vb.major ? -1 : 1;
  }
  if (va.minor !== vb.minor) {
    return va.minor < vb.minor ? -1 : 1;
  }
  if (va.patch !== vb.patch) {
    return va.patch < vb.patch ? -1 : 1;
  }

  // Compare prerelease
  var aHasPre = va.prereleaseIdentifiers !== null;
  var bHasPre = vb.prereleaseIdentifiers !== null;

  if (!aHasPre && !bHasPre) {
    return 0;
  }

  // A version with a prerelease has lower precedence than the same version without one
  if (!aHasPre && bHasPre) {
    return 1;
  }
  if (aHasPre && !bHasPre) {
    return -1;
  }

  // Both have prerelease identifiers
  var aIds = va.prereleaseIdentifiers;
  var bIds = vb.prereleaseIdentifiers;
  var minLen = Math.min(aIds.length, bIds.length);

  for (var i = 0; i < minLen; i++) {
    var cmp = comparePrereleaseIdentifiers(aIds[i], bIds[i]);
    if (cmp !== 0) {
      return cmp;
    }
  }

  // All preceding identifiers equal: fewer identifiers is lower
  if (aIds.length < bIds.length) {
    return -1;
  }
  if (aIds.length > bIds.length) {
    return 1;
  }

  return 0;
}

module.exports = {
  compareSemver: compareSemver
};
