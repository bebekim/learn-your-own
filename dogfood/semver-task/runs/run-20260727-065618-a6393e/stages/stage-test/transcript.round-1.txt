// Normalize -0 to +0 so Object.is comparisons treat them as equal.
const num = (x) => x + 0;

test('basic major/minor/patch numeric comparison', () => {
  assert.equal(num(compareSemver('1.2.3', '1.2.4')), -1);

});

});

test('minor compared numerically when major equal', () => {
  assert.equal(num(compareSemver('1.2.0', '1.10.0')), -1);
  assert.equal(num(compareSemver('1.10.0', '1.2.0')), 1);

test('patch compared numerically when major and minor equal', () => {
  assert.equal(num(compareSemver('1.0.2', '1.0.10')), -1);
  assert.equal(num(compareSemver('1.0.10', '1.0.2')), 1);

test('mixed numeric and alphanumeric prerelease identifiers', () => {
  // numeric 1 < 'beta' (alphanumeric)
  assert.equal(num(compareSemver('1.0.0-1', '1.0.0-beta')), -1);
  assert.equal(num(compareSemver('1.0.0-beta', '1.0.0-1')), 1);

  assert.equal(num(compareSemver('1.0.0-alpha', '1.0.0-beta')), -1);

  assert.equal(num(compareSemver('1.0.0-99', '1.0.0-a')), -1);

test('complete precedence ordering example', () => {
  // Classic semver precedence chain
  const chain = [
    '1.0.0-alpha',
    '1.0.0-alpha.1',
    '1.0.0-alpha.beta',
    '1.0.0-beta',
    '1.0.0-beta.2',
    '1.0.0-beta.11',
    '1.0.0-rc.1',
    '1.0.0',
  ];
  for (let i = 0; i < chain.length; i++) {
    for (let j = 0; j < chain.length; j++) {
      const expected = i < j ? -1 : i > j ? 1 : 0;
      assert.equal(num(compareSemver(chain[i], chain[j])), expected);
    }
  }
});

test('zero versions compare correctly', () => {
