const parseCsvLine = (line) => {
  if (line === '') {
    return [''];
  }

  const fields = [];
  let current = '';
  let inQuotes = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];

    if (!inQuotes) {
      if (char === ',') {
        fields.push(current);
        current = '';
      } else if (char === '"') {
        inQuotes = true;
      } else {
        current += char;
      }
    } else {
      if (char === '"') {
        if (i + 1 < line.length && line[i + 1] === '"') {
          current += '"';
          i++;
        } else {
          inQuotes = false;
        }
      } else {
        current += char;
      }
    }
  }

  fields.push(current);
  return fields;
};

module.exports = { parseCsvLine };
