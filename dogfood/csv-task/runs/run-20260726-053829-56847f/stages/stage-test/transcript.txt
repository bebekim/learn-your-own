import { parseCsvLine } from '../src/csv-line.js';

test('empty string returns a single empty field', () => {
  expect(parseCsvLine('')).toEqual(['']);
});

test('fields split on commas outside quotes', () => {
  expect(parseCsvLine('a,b,c')).toEqual(['a', 'b', 'c']);
});

test('quoted field removes surrounding quotes', () => {
  expect(parseCsvLine('"hello, world",x')).toEqual(['hello, world', 'x']);
});

test('quoted field preserves internal commas', () => {
  expect(parseCsvLine('"she said \"\"hi\"\"\",y')).toEqual(['she said "hi"', 'y']);
});

test('two double quotes inside quoted field decode to one', () => {
  expect(parseCsvLine('"a""b",c')).toEqual(['a"b', 'c']);
});

test('empty fields between consecutive commas', () => {
  expect(parseCsvLine('a,,b')).toEqual(['a', '', 'b']);
});

test('quoted empty string', () => {
  expect(parseCsvLine('""')).toEqual(['']);
});

test('field with only quotes', () => {
  expect(parseCsvLine('"x"')).toEqual(['x']);
});

test('field with quotes and content', () => {
  expect(parseCsvLine('"x"')).toEqual(['x']);
});

test('quoted field with leading/trailing spaces preserved', () => {
  expect(parseCsvLine('" x "')).toEqual([' x ']);
});

test('field with leading/trailing spaces outside quotes preserved', () => {
  expect(parseCsvLine(' a , b ')).toEqual([' a ', ' b ']);
});
